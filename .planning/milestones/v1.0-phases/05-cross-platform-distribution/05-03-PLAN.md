Create `.github/workflows/publish.yml` with the following structure:

```yaml
name: publish
on:
  push:
    tags:
      - 'v*'

jobs:
  publish-tauri:
    permissions:
      contents: write
    strategy:
      fail-fast: false
      matrix:
        include:
          - platform: 'macos-latest'
            args: '--target universal-apple-darwin'
          - platform: 'ubuntu-22.04'
            args: ''
          - platform: 'windows-latest'
            args: ''
    runs-on: ${{ matrix.platform }}
    steps:
      - uses: actions/checkout@v4

      - name: Install dependencies (Ubuntu only)
        if: matrix.platform == 'ubuntu-22.04'
        run: |
          sudo apt-get update
          sudo apt-get install -y libwebkit2gtk-4.1-dev libappindicator3-dev librsvg2-dev patchelf

      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: lts/*
          cache: 'npm'

      - name: Install Rust stable
        uses: dtolnay/rust-toolchain@stable
        with:
          targets: ${{ matrix.platform == 'macos-latest' && 'aarch64-apple-darwin,x86_64-apple-darwin' || '' }}

      - name: Rust cache
        uses: swatinem/rust-cache@v2
        with:
          workspaces: './src-tauri -> target'

      - name: Install frontend dependencies
        run: npm ci

      - name: Run tests
        run: npm test && cargo test --manifest-path src-tauri/Cargo.toml

      - uses: tauri-apps/tauri-action@v1
        id: tauri-build
        env:
          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}
          TAURI_SIGNING_PRIVATE_KEY: ${{ secrets.TAURI_SIGNING_PRIVATE_KEY }}
          TAURI_SIGNING_PRIVATE_KEY_PASSWORD: ${{ secrets.TAURI_SIGNING_PRIVATE_KEY_PASSWORD }}
        with:
          tagName: v__VERSION__
          releaseName: 'Nexus v__VERSION__'
          releaseBody: 'See Assets to download and install.'
          releaseDraft: true
          prerelease: false
          uploadUpdaterJson: true
          args: ${{ matrix.args }}

      - name: Check artifact sizes (must be under 15 MB)
        shell: bash
        run: |
          MAX_BYTES=15728640  # 15 MB
          OVERSIZED=0
          echo "Checking release artifact sizes (max 15 MB)..."
          for artifact in src-tauri/target/release/bundle/**/*; do
            if [ -f "$artifact" ]; then
              SIZE=$(stat -f%z "$artifact" 2>/dev/null || stat -c%s "$artifact" 2>/dev/null || echo 0)
              SIZE_MB=$(echo "scale=2; $SIZE / 1048576" | bc 2>/dev/null || echo "unknown")
              BASENAME=$(basename "$artifact")
              echo "  $BASENAME: ${SIZE_MB} MB ($SIZE bytes)"
              if [ "$SIZE" -gt "$MAX_BYTES" ]; then
                echo "::error::Artifact $BASENAME exceeds 15 MB limit: ${SIZE_MB} MB"
                OVERSIZED=1
              fi
            fi
          done
          if [ "$OVERSIZED" -eq 1 ]; then
            echo "::error::One or more artifacts exceed the 15 MB size limit. Review release profile optimizations."
            exit 1
          fi
          echo "All artifacts within 15 MB limit."
```

Key design decisions:
- `fail-fast: false` so one platform failing doesn't cancel others
- `npm ci` instead of `npm install` for reproducible builds
- Tests run before build to catch issues early
- macOS runner installs both aarch64 and x86_64 Rust targets for universal binary
- Ubuntu installs all required system deps (webkit2gtk, appindicator, librsvg2, patchelf)
- `uploadUpdaterJson: true` generates the `latest.json` manifest for the auto-updater
- `releaseDraft: true` per user decision — user reviews and publishes manually
- `tagName: v__VERSION__` uses the version from tauri.conf.json
- Post-build size-check step scans all bundle artifacts and fails the job if any exceeds 15 MB (15728640 bytes). Uses `stat` with fallback for cross-platform compatibility (macOS uses `-f%z`, Linux uses `-c%s`). This enforces PERF-04 in CI, not just via release profile settings.
